-- [SQLite 전용] strftime('%Y-%m', …). MySQL: DATE_FORMAT(created_at, '%Y-%m'), PostgreSQL: to_char(created_at, 'YYYY-MM')
sqlite> SELECT strftime('%Y-%m', created_at) AS month,
   ...>        COUNT(*)                      AS review_count,
   ...>        ROUND(AVG(rating), 2)         AS avg_rating
   ...> FROM review
   ...> GROUP BY month
   ...> ORDER BY month;
